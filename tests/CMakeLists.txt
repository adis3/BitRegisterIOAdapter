cmake_minimum_required(VERSION 3.16)
project(BitRegisterIOAdapterTests LANGUAGES CXX)
add_executable(adapter_test adapter_test.cpp ../src/BitRegisterIOAdapter.cpp)
target_compile_features(adapter_test PRIVATE cxx_std_11)
target_include_directories(adapter_test PRIVATE stubs ../src ../../ILogger/src)
if(MSVC)
  target_compile_options(adapter_test PRIVATE /W4 /WX /EHsc)
  target_compile_definitions(adapter_test PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(adapter_test PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME adapter_behavior COMMAND adapter_test)
